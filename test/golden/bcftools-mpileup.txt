Name:  bcftools-mpileup-input


Desc:  bcftools-mpileup-input


Usage:
bcftools mpileup [options] in1.bam [in2.bam [...]]


(bcftools-mpileup-input) -6 --illumina1.3+  ::  
quality is in the Illumina-1.3+ encoding

(bcftools-mpileup-input) -A --count-orphans  ::  
do not discard anomalous read pairs

(bcftools-mpileup-input) -b --bam-list  ::  FILE
list of input BAM filenames, one per line

(bcftools-mpileup-input) -B --no-BAQ  ::  
disable BAQ (per-Base Alignment Quality)

(bcftools-mpileup-input) -C --adjust-MQ  ::  INT
adjust mapping quality; recommended:50, disable:0 [0]

(bcftools-mpileup-input) -d --max-depth  ::  INT
max raw per-file depth; avoids excessive memory usage [250]

(bcftools-mpileup-input) -E --redo-BAQ  ::  
recalculate BAQ on the fly, ignore existing BQs

(bcftools-mpileup-input) -f --fasta-ref  ::  FILE
faidx indexed reference sequence file

(bcftools-mpileup-input) --no-reference  ::  
do not require fasta reference file

(bcftools-mpileup-input) -G --read-groups  ::  FILE
select or exclude read groups listed in the file

(bcftools-mpileup-input) -q --min-MQ  ::  INT
skip alignments with mapQ smaller than INT [0]

(bcftools-mpileup-input) -Q --min-BQ  ::  INT
skip bases with baseQ/BAQ smaller than INT [13]

(bcftools-mpileup-input) -r --regions  ::  REG
comma separated list of regions in which pileup is generated

(bcftools-mpileup-input) -R --regions-file  ::  FILE
restrict to regions listed in a file

(bcftools-mpileup-input) --ignore-RG  ::  
ignore RG tags (one BAM = one sample)

(bcftools-mpileup-input) --rf --incl-flags  ::  STR|INT
required flags: skip reads with mask bits unset []

(bcftools-mpileup-input) --ff --excl-flags  ::  STR|INT
filter flags: skip reads with mask bits set

(bcftools-mpileup-input) -s --samples  ::  LIST
comma separated list of samples to include

(bcftools-mpileup-input) -S --samples-file  ::  FILE
file of samples to include

(bcftools-mpileup-input) -t --targets  ::  REG
similar to -r but streams rather than index-jumps

(bcftools-mpileup-input) -T --targets-file  ::  FILE
similar to -R but streams rather than index-jumps

(bcftools-mpileup-input) -x --ignore-overlaps  ::  
disable read-pair overlap detection

(bcftools-mpileup-input) -a --annotate  ::  LIST
optional tags to output; '?' to list []

(bcftools-mpileup-input) -g --gvcf  ::  INT[,...]
group non-variant sites into gVCF blocks according to minimum per-sample DP

(bcftools-mpileup-input) --no-version  ::  
do not append version and command line to the header

(bcftools-mpileup-input) -o --output  ::  FILE
write output to FILE [standard output]

(bcftools-mpileup-input) -O --output-type  ::  TYPE
'b' compressed BCF; 'u' uncompressed BCF; 'z' compressed VCF; 'v' uncompressed VCF [v]

(bcftools-mpileup-input) --threads  ::  INT
use multithreading with INT worker threads [0]

(bcftools-mpileup-input) -e --ext-prob  ::  INT
Phred-scaled gap extension seq error probability [20]

(bcftools-mpileup-input) -F --gap-frac  ::  FLOAT
minimum fraction of gapped reads [0.002]

(bcftools-mpileup-input) -h --tandem-qual  ::  INT
coefficient for homopolymer errors [100]

(bcftools-mpileup-input) -I --skip-indels  ::  
do not perform indel calling

(bcftools-mpileup-input) -L --max-idepth  ::  INT
maximum per-file depth for INDEL calling [250]

(bcftools-mpileup-input) -m --min-ireads  ::  INT
minimum number gapped reads for indel candidates [1]

(bcftools-mpileup-input) -o --open-prob  ::  INT
Phred-scaled gap open seq error probability [40]

(bcftools-mpileup-input) -p --per-sample-mF  ::  
apply -m and -F per-sample for increased sensitivity

(bcftools-mpileup-input) -P --platforms  ::  STR
comma separated list of platforms for indels [all]
