Name:  rsync-input


Desc:  rsync-input


Usage:
rsync [OPTION]... SRC [SRC]... DEST


(rsync-input) --verbose -v  ::  
increase verbosity

(rsync-input) --info  ::  FLAGS
fine-grained informational verbosity

(rsync-input) --debug  ::  FLAGS
fine-grained debug verbosity

(rsync-input) --stderr  ::  e|a|c
change stderr output mode (default: errors)

(rsync-input) --quiet -q  ::  
suppress non-error messages

(rsync-input) --no-motd  ::  
suppress daemon-mode MOTD

(rsync-input) --checksum -c  ::  
skip based on checksum, not mod-time & size

(rsync-input) --archive -a  ::  
archive mode; equals -rlptgoD (no -H,-A,-X)

(rsync-input) --no-OPTION  ::  
turn off an implied OPTION (e.g. --no-D)

(rsync-input) --recursive -r  ::  
recurse into directories

(rsync-input) --relative -R  ::  
use relative path names

(rsync-input) --no-implied-dirs  ::  
don't send implied dirs with --relative

(rsync-input) --backup -b  ::  
make backups (see --suffix & --backup-dir)

(rsync-input) --backup-dir  ::  DIR
make backups into hierarchy based in DIR

(rsync-input) --suffix  ::  SUFFIX
backup suffix (default ~ w/o --backup-dir)

(rsync-input) --update -u  ::  
skip files that are newer on the receiver

(rsync-input) --inplace  ::  
update destination files in-place

(rsync-input) --append  ::  
append data onto shorter files

(rsync-input) --append-verify  ::  
--append w/old data in file checksum

(rsync-input) --dirs -d  ::  
transfer directories without recursing

(rsync-input) --mkpath  ::  
create the destination's path component

(rsync-input) --links -l  ::  
copy symlinks as symlinks

(rsync-input) --copy-links -L  ::  
transform symlink into referent file/dir

(rsync-input) --copy-unsafe-links  ::  
only "unsafe" symlinks are transformed

(rsync-input) --safe-links  ::  
ignore symlinks that point outside the tree

(rsync-input) --munge-links  ::  
munge symlinks to make them safe & unusable

(rsync-input) --copy-dirlinks -k  ::  
transform symlink to dir into referent dir

(rsync-input) --keep-dirlinks -K  ::  
treat symlinked dir on receiver as dir

(rsync-input) --hard-links -H  ::  
preserve hard links

(rsync-input) --perms -p  ::  
preserve permissions

(rsync-input) --executability -E  ::  
preserve executability

(rsync-input) --chmod  ::  CHMOD
affect file and/or directory permissions

(rsync-input) --acls -A  ::  
preserve ACLs (implies --perms)

(rsync-input) --xattrs -X  ::  
preserve extended attributes

(rsync-input) --owner -o  ::  
preserve owner (super-user only)

(rsync-input) --group -g  ::  
preserve group

(rsync-input) --devices  ::  
preserve device files (super-user only)

(rsync-input) --specials  ::  
preserve special files

(rsync-input) -D  ::  
same as --devices --specials

(rsync-input) --times -t  ::  
preserve modification times

(rsync-input) --atimes -U  ::  
preserve access (use) times

(rsync-input) --open-noatime  ::  
avoid changing the atime on opened files

(rsync-input) --crtimes -N  ::  
preserve create times (newness)

(rsync-input) --omit-dir-times -O  ::  
omit directories from --times

(rsync-input) --omit-link-times -J  ::  
omit symlinks from --times

(rsync-input) --super  ::  
receiver attempts super-user activities

(rsync-input) --fake-super  ::  
store/recover privileged attrs using xattrs

(rsync-input) --sparse -S  ::  
turn sequences of nulls into sparse blocks

(rsync-input) --preallocate  ::  
allocate dest files before writing them

(rsync-input) --write-devices  ::  
write to devices as files (implies --inplace)

(rsync-input) --dry-run -n  ::  
perform a trial run with no changes made

(rsync-input) --whole-file -W  ::  
copy files whole (w/o delta-xfer algorithm)

(rsync-input) --checksum-choice  ::  STR
choose the checksum algorithm (aka --cc)

(rsync-input) --one-file-system -x  ::  
don't cross filesystem boundaries

(rsync-input) --block-size -B  ::  SIZE
force a fixed checksum block-size

(rsync-input) --rsh -e  ::  COMMAND
specify the remote shell to use

(rsync-input) --rsync-path  ::  PROGRAM
specify the rsync to run on remote machine

(rsync-input) --existing  ::  
skip creating new files on receiver

(rsync-input) --ignore-existing  ::  
skip updating files that exist on receiver

(rsync-input) --remove-source-files  ::  
sender removes synchronized files (non-dir)

(rsync-input) --del  ::  
an alias for --delete-during

(rsync-input) --delete  ::  
delete extraneous files from dest dirs

(rsync-input) --delete-before  ::  
receiver deletes before xfer, not during

(rsync-input) --delete-during  ::  
receiver deletes during the transfer

(rsync-input) --delete-delay  ::  
find deletions during, delete after

(rsync-input) --delete-after  ::  
receiver deletes after transfer, not during

(rsync-input) --delete-excluded  ::  
also delete excluded files from dest dirs

(rsync-input) --ignore-missing-args  ::  
ignore missing source args without error

(rsync-input) --delete-missing-args  ::  
delete missing source args from destination

(rsync-input) --ignore-errors  ::  
delete even if there are I/O errors

(rsync-input) --force  ::  
force deletion of dirs even if not empty

(rsync-input) --max-delete  ::  NUM
don't delete more than NUM files

(rsync-input) --max-size  ::  SIZE
don't transfer any file larger than SIZE

(rsync-input) --min-size  ::  SIZE
don't transfer any file smaller than SIZE

(rsync-input) --max-alloc  ::  SIZE
change a limit relating to memory alloc

(rsync-input) --partial  ::  
keep partially transferred files

(rsync-input) --partial-dir  ::  DIR
put a partially transferred file into DIR

(rsync-input) --delay-updates  ::  
put all updated files into place at end

(rsync-input) --prune-empty-dirs -m  ::  
prune empty directory chains from file-list

(rsync-input) --numeric-ids  ::  
don't map uid/gid values by user/group name

(rsync-input) --usermap  ::  STRING
custom username mapping

(rsync-input) --groupmap  ::  STRING
custom groupname mapping

(rsync-input) --chown  ::  USER:GROUP
simple username/groupname mapping

(rsync-input) --timeout  ::  SECONDS
set I/O timeout in seconds

(rsync-input) --contimeout  ::  SECONDS
set daemon connection timeout in seconds

(rsync-input) --ignore-times -I  ::  
don't skip files that match size and time

(rsync-input) --size-only  ::  
skip files that match in size

(rsync-input) --modify-window -@  ::  NUM
set the accuracy for mod-time comparisons

(rsync-input) --temp-dir -T  ::  DIR
create temporary files in directory DIR

(rsync-input) --fuzzy -y  ::  
find similar file for basis if no dest file

(rsync-input) --compare-dest  ::  DIR
also compare destination files relative to DIR

(rsync-input) --copy-dest  ::  DIR
... and include copies of unchanged files

(rsync-input) --link-dest  ::  DIR
hardlink to files in DIR when unchanged

(rsync-input) --compress -z  ::  
compress file data during the transfer

(rsync-input) --compress-choice  ::  STR
choose the compression algorithm (aka --zc)

(rsync-input) --compress-level  ::  NUM
explicitly set compression level (aka --zl)

(rsync-input) --skip-compress  ::  LIST
skip compressing files with suffix in LIST

(rsync-input) --cvs-exclude -C  ::  
auto-ignore files in the same way CVS does

(rsync-input) --filter -f  ::  RULE
add a file-filtering RULE

(rsync-input) -F  ::  
same as --filter='dir-merge /.rsync-filter' repeated: --filter='- .rsync-filter'

(rsync-input) --exclude  ::  PATTERN
exclude files matching PATTERN

(rsync-input) --exclude-from  ::  FILE
read exclude patterns from FILE

(rsync-input) --include  ::  PATTERN
don't exclude files matching PATTERN

(rsync-input) --include-from  ::  FILE
read include patterns from FILE

(rsync-input) --files-from  ::  FILE
read list of source-file names from FILE

(rsync-input) --from0 -0  ::  
all *-from/filter files are delimited by 0s

(rsync-input) --protect-args -s  ::  
no space-splitting; wildcard chars only

(rsync-input) --copy-as  ::  USER[:GROUP]
specify user & optional group for the copy

(rsync-input) --address  ::  ADDRESS
bind address for outgoing socket to daemon

(rsync-input) --port  ::  PORT
specify double-colon alternate port number

(rsync-input) --sockopts  ::  OPTIONS
specify custom TCP options

(rsync-input) --blocking-io  ::  
use blocking I/O for the remote shell

(rsync-input) --outbuf  ::  N|L|B
set out buffering to None, Line, or Block

(rsync-input) --stats  ::  
give some file-transfer stats

(rsync-input) --8-bit-output -8  ::  
leave high-bit chars unescaped in output

(rsync-input) --human-readable -h  ::  
output numbers in a human-readable format

(rsync-input) --progress  ::  
show progress during transfer

(rsync-input) -P  ::  
same as --partial --progress

(rsync-input) --itemize-changes -i  ::  
output a change-summary for all updates

(rsync-input) --remote-option -M  ::  OPT
send OPTION to the remote side only

(rsync-input) --out-format  ::  FORMAT
output updates using the specified FORMAT

(rsync-input) --log-file  ::  FILE
log what we're doing to the specified FILE

(rsync-input) --log-file-format  ::  FMT
log updates using the specified FMT

(rsync-input) --password-file  ::  FILE
read daemon-access password from FILE

(rsync-input) --early-input  ::  FILE
use FILE for daemon's early exec input

(rsync-input) --list-only  ::  
list the files instead of copying them

(rsync-input) --bwlimit  ::  RATE
limit socket I/O bandwidth

(rsync-input) --stop-after  ::  MINS
Stop rsync after MINS minutes have elapsed

(rsync-input) --stop-at  ::  y-m-dTh:m
Stop rsync at the specified point in time

(rsync-input) --write-batch  ::  FILE
write a batched update to FILE

(rsync-input) --only-write-batch  ::  FILE
like --write-batch but w/o updating dest

(rsync-input) --read-batch  ::  FILE
read a batched update from FILE

(rsync-input) --protocol  ::  NUM
force an older protocol version to be used

(rsync-input) --iconv  ::  CONVERT_SPEC
request charset conversion of filenames

(rsync-input) --checksum-seed  ::  NUM
set block/file checksum seed (advanced)

(rsync-input) --ipv4 -4  ::  
prefer IPv4

(rsync-input) --ipv6 -6  ::  
prefer IPv6

(rsync-input) --version -V  ::  
print the version + other info and exit

(rsync-input) --help -h  ::  (*)
show this help (* -h is help only on its own)
